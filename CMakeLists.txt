cmake_minimum_required(VERSION 3.16)
project(CurveFitter VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Qml)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Qml)
add_library(curvefitter src/curvefitter.cpp src/minimizer.cpp src/customformula.cpp)
target_include_directories(curvefitter PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(curvefitter PUBLIC Threads::Threads Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Qml)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
    set(CMAKE_AUTOMOC ON)
    add_executable(curvefitter_tests tests/test_curvefitter.cpp)
    target_link_libraries(curvefitter_tests PRIVATE curvefitter Qt${QT_VERSION_MAJOR}::Test)
    target_compile_definitions(curvefitter_tests PRIVATE REFERENCE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/tests/java_reference.jsonl")
    add_test(NAME curvefitter_tests COMMAND curvefitter_tests)
endif()
add_executable(curvefitter_example examples/main.cpp)
target_link_libraries(curvefitter_example PRIVATE curvefitter)

option(CURVEFITTER_BUILD_WIDGETS "Build native Qt settings and plot widgets" ON)
if(CURVEFITTER_BUILD_WIDGETS)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
    add_library(curvefitter_widgets src/curvefitterwidgets.cpp)
    target_link_libraries(curvefitter_widgets PUBLIC curvefitter Qt${QT_VERSION_MAJOR}::Widgets)
    if(BUILD_TESTING)
        add_executable(curvefitter_widgets_tests tests/test_widgets.cpp)
        target_link_libraries(curvefitter_widgets_tests PRIVATE curvefitter_widgets Qt${QT_VERSION_MAJOR}::Test)
        add_test(NAME curvefitter_widgets_tests COMMAND curvefitter_widgets_tests)
        set_tests_properties(curvefitter_widgets_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
endif()
